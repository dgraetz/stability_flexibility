Multilevel Test of Stability-Flexibility

This script cleans and analyses the AlternateRuns Dataset.

Variables - block: 0 for practice, there are a total of 8 blocks per part and each block consists of 112 trials. So 896 trials per participant over all 7 blocks (not counting practice)

  • bal: counterbalancing of conditions across blocks

  • x, y, c2: irrelevant (already taken out)

  • cycle: counting within full alternating cycle (8), switch at 1 and 5

  • task: 1=shape, 2=color

  • dimshape=specific shapes–4=neutral

  • dimcolor=specific color–4=neutral

  • correct: correct response (i.e., value of the currently relevant task dimension)

  • error: 0 = no error, 1 = yes error

  • response: actual response

  • RT: or response time

alt_run <- alt_run %>% 
  arrange(id, trial) %>% 
  group_by(id, block) %>%
  mutate(
    switch = if_else(cycle %in% c(1,5), 1, 0),
    cycle2 = ifelse(cycle > 4, cycle - 4, cycle),
    group = ifelse(cycle2 == 1, 1, 0),
    group = cumsum(group),
    lerror = lag(error, 1), 
    ltrial = lag(trial, 1), 
    lrt = lag(rt, 1),
    valence = if_else(dimshape == 4 | dimcolor == 4, 1, 2),  # 1 = uni, 2 = bi
    lvalence = lag(valence, 1)                               # previous trial’s valence
  ) %>%
  group_by(id, block, group) %>%
  mutate(lvalence = lvalence[1]) %>%                         # constant within 4‑trial group
  ungroup()


alt_run <- alt_run %>% #making contrasts outlined in prereg
  group_by(id, block, group) %>% 
  mutate(
    lvalence_contr = case_when(lvalence == 1 ~ -0.5, lvalence == 2 ~ 0.5),
    valence_contr = case_when(valence == 1 ~ -0.5, valence == 2 ~ 0.5)
    )

#function for CI

norm_for_errorbars <- function(dv,ID){
  dat <- tibble(dv = dv,ID=ID)
  dat2 <- dat %>%
    mutate(grand_mean = mean(dv,na.rm=T)) %>%
    group_by(ID) %>%
    mutate(submean = mean(dv,na.rm=T)) %>%
    ungroup() %>%
    mutate(normdv = dv - submean + grand_mean)
  return(dat2$normdv)
}
alt_run <- alt_run %>% #norm on trial level
  group_by(id) %>%
  mutate(
    rt_norm       = norm_for_errorbars(rt, id),
    rt_log_norm   = norm_for_errorbars(rt_log, id),
    error_norm    = norm_for_errorbars(error, id)
  ) %>%
  ungroup()


allagg <- alt_run %>%
  group_by(id, cycle2, switch, valence, lvalence) %>% #first agg
  summarize(
    rt_mean          = mean(rt[filter_correct == 0], na.rm = TRUE),
    rt_log_mean      = mean(rt_log[filter_correct == 0], na.rm = TRUE),
    error_mean       = mean(error, na.rm = TRUE),

    rt_norm_mean     = mean(rt_norm[filter_correct == 0], na.rm = TRUE),
    rt_log_norm_mean = mean(rt_log_norm[filter_correct == 0], na.rm = TRUE),
    error_norm_mean  = mean(error_norm, na.rm = TRUE),

    .groups = "drop"
  )


allagg_ci <- allagg %>% #group level CIs
  group_by(cycle2, switch, valence, lvalence) %>%
  summarize(
    n = n(),

    rt_mean      = mean(rt_mean),    # grand means
    rt_log_mean  = mean(rt_log_mean),
    error_mean   = mean(error_mean),

    rt_se        = sd(rt_norm_mean) / sqrt(n),    # SEs using normalized participant means
    rt_log_se    = sd(rt_log_norm_mean) / sqrt(n),
    error_se     = sd(error_norm_mean) / sqrt(n),

    rt_ci_low        = rt_mean - 1.96 * rt_se,    # 95% 
    rt_ci_high       = rt_mean + 1.96 * rt_se,
    rt_log_ci_low    = rt_log_mean - 1.96 * rt_log_se,
    rt_log_ci_high   = rt_log_mean + 1.96 * rt_log_se,
    error_ci_low     = error_mean - 1.96 * error_se,
    error_ci_high    = error_mean + 1.96 * error_se,

    .groups = "drop"
  ) %>%
  mutate(
    valence = factor(valence, #make it nice for plot
                     levels = c(1, 2),
                     labels = c("Univalent", "Bivalent")),
    lvalence = factor(lvalence,
                      levels = c(1, 2),
                      labels = c("Univalent", "Bivalent")),
    cycle2 = factor(cycle2)
  )
